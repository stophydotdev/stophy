```bash cURL
curl -X POST https://api.stophy.dev/v1/parse/upload-url \
  -H "Authorization: Bearer $STOPHY_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{"filename":"report.pdf","contentType":"application/pdf","declaredSizeBytes":1048576}'
```
```typescript TypeScript
const response = await fetch("https://api.stophy.dev/v1/parse/upload-url", {
  method: "POST",
  headers: {
    Authorization: `Bearer ${process.env.STOPHY_API_KEY}`,
    "Content-Type": "application/json",
  },
  body: JSON.stringify({
    filename: "report.pdf",
    contentType: "application/pdf",
    declaredSizeBytes: 1048576,
  }),
});
console.log(await response.json());
```
```python Python
import os

import requests

response = requests.post(
    "https://api.stophy.dev/v1/parse/upload-url",
    headers={"Authorization": f"Bearer {os.environ['STOPHY_API_KEY']}"},
    json={
        "filename": "report.pdf",
        "contentType": "application/pdf",
        "declaredSizeBytes": 1048576,
    },
)
print(response.json())
```
