```typescript TypeScript
const response = await fetch("https://api.stophy.dev/v1/feedback", {
  method: "POST",
  headers: {
    Authorization: `Bearer ${process.env.STOPHY_API_KEY}`,
    "Content-Type": "application/json",
  },
  body: JSON.stringify({
    rating: "good",
    endpoint: "scrape",
    jobId: "ac28ed59-d0d8-4639-92e1-e1b0db5e9abb",
  }),
});
console.log(await response.json());
```
```python Python
import os

import requests

response = requests.post(
    "https://api.stophy.dev/v1/feedback",
    headers={"Authorization": f"Bearer {os.environ['STOPHY_API_KEY']}"},
    json={
        "rating": "good",
        "endpoint": "scrape",
        "jobId": "ac28ed59-d0d8-4639-92e1-e1b0db5e9abb",
    },
)
print(response.json())
```
