```bash cURL
curl -X POST https://api.stophy.dev/v1/browser/network-policies \
  -H "Authorization: Bearer $STOPHY_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{"name":"docs-example","defaultAction":"block","allow":{"domains":["bun.com"]}}'
```
```typescript TypeScript
const response = await fetch("https://api.stophy.dev/v1/browser/network-policies", {
  method: "POST",
  headers: {
    Authorization: `Bearer ${process.env.STOPHY_API_KEY}`,
    "Content-Type": "application/json",
  },
  body: JSON.stringify({
    name: "docs-example",
    defaultAction: "block",
    allow: { domains: ["bun.com"] },
  }),
});
console.log(await response.json());
```
```python Python
import os

import requests

response = requests.post(
    "https://api.stophy.dev/v1/browser/network-policies",
    headers={"Authorization": f"Bearer {os.environ['STOPHY_API_KEY']}"},
    json={
        "name": "docs-example",
        "defaultAction": "block",
        "allow": {"domains": ["bun.com"]},
    },
)
print(response.json())
```
