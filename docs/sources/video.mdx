```bash cURL
curl -X POST https://api.stophy.dev/v1/scrape \
  -H "Authorization: Bearer $STOPHY_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{"url":"https://www.youtube.com/results?search_query=lofi+beats","formats":[{"type":"json","parser":"youtube-search"}]}'
```
