```bash cURL
  -H "Authorization: Bearer $STOPHY_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{"url":"https://www.google.com/maps/search/dentists+in+Ljubljana","formats":[{"type":"json","parser":"google-maps"}]}'
```
